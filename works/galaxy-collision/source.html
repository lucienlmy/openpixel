function mkGal(s,core,nrm,spin,n,tag,r){// a thin disk with two loose arms, stars on circular orbits around the core
  const up=Math.abs(nrm[2])<0.9?[0,0,1]:[1,0,0];const u=norm(cross(up,nrm)),v=cross(nrm,u);
